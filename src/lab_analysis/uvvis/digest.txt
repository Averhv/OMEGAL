from __future__ import annotations

import argparse
import sys

from . import config
from .runner import run


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(
        prog="uvvis-analysis",
        description="UV-Vis spectroscopy and chlorophyll pigment analysis.",
    )
    parser.add_argument("input_dir", nargs="?", default=None,
                        help="Folder with .txt/.csv spectral exports (default: cwd).")
    parser.add_argument("--output-dir", default=None,
                        help="Where to write results. Default: <input_dir>/uvvis_results.")
    parser.add_argument("--solvent", choices=["ethanol", "acetone", "methanol"], default="ethanol")
    args = parser.parse_args(argv)

    if args.solvent:
        config.SOLVENT = args.solvent

    try:
        run(input_dir=args.input_dir, output_dir=args.output_dir)
    except Exception as exc:
        print(f"[ERROR] {exc}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Optional
import numpy as np

from . import config


@dataclass
class Spectrum:
    """One UV-Vis spectral measurement: wavelength and absorbance arrays + metadata."""
    name: str
    source: str
    wavelength: np.ndarray
    absorbance: np.ndarray
    stats: dict = field(default_factory=dict)
    group: str = ""
    replicate: str = ""
    sample: str = ""
    dilution_factor: float = config.DEFAULT_DILUTION_FACTOR
    extract_volume_ml: float = config.DEFAULT_EXTRACT_VOLUME_ML
    dry_mass_mg: float = config.DEFAULT_DRY_MASS_MG
    flags: list = field(default_factory=list)

    @property
    def label(self) -> str:
        return config.CUSTOM_LEGEND_LABELS.get(self.name, self.name)

    def get_absorbance_at(self, wl: float) -> float:
        """Linear interpolation of absorbance at a given wavelength (nm)."""
        if self.wavelength.size == 0:
            return 0.0
        return float(np.interp(wl, self.wavelength, self.absorbance, left=np.nan, right=np.nan))


@dataclass
class Condition:
    """A set of replicate spectral measurements sharing sample/condition."""
    name: str
    members: list[Spectrum]
    sample: str
    dilution_factor: float
    extract_volume_ml: float
    dry_mass_mg: float

from __future__ import annotations

import re
from pathlib import Path
import numpy as np

from .models import Spectrum
from ..common.io import read_text


def split_replicate(name: str) -> tuple[str, str]:
    stem = Path(name).stem
    match = re.search(r"[-_](?:rep)?([0-9]+|[a-zA-Z])$", stem)
    if match:
        rep = match.group(1)
        base = stem[:match.start()]
        return base.strip("-_"), rep
    return stem, ""


def parse_spectrum_file(path: Path) -> Spectrum:
    path = Path(path)
    text = read_text(path)
    lines = text.splitlines()

    wavelengths = []
    absorbances = []

    for line in lines:
        line = line.strip()
        if not line or line.startswith("#"):
            continue
        cleaned = line.replace('"', '').replace("'", "")
        for sep in (';', ',', '\t'):
            parts = [p.strip() for p in cleaned.split(sep) if p.strip()]
            if len(parts) >= 2:
                try:
                    wl = float(parts[0].replace(',', '.'))
                    abs_val = float(parts[1].replace(',', '.'))
                    wavelengths.append(wl)
                    absorbances.append(abs_val)
                    break
                except ValueError:
                    continue

    if not wavelengths:
        raise ValueError(f"Could not parse valid wavelength/absorbance data from {path.name}")

    wl_arr = np.array(wavelengths, dtype=float)
    abs_arr = np.array(absorbances, dtype=float)

    sort_idx = np.argsort(wl_arr)
    wl_arr = wl_arr[sort_idx]
    abs_arr = abs_arr[sort_idx]

    base, rep = split_replicate(path.name)
    spec = Spectrum(
        name=path.stem,
        source=path.name,
        wavelength=wl_arr,
        absorbance=abs_arr,
        group=base,
        replicate=rep,
        sample=base,
    )
    return spec

@dataclass
class Curve:
    x: np.ndarray
    y: np.ndarray
    label: str
    color: str
    lw: float = 2.1
    alpha: float = 1.0
    y_sd: Optional[np.ndarray] = None
    legend: bool = True


def x_limits(curves: list[Curve]) -> tuple[float, float]:
    if config.X_LIMITS:
        return config.X_LIMITS
    if config.X_RANGE_MODE.lower() == "full":
        return min(c.x.min() for c in curves), max(c.x.max() for c in curves)
    lo = min(c.x.min() for c in curves)
    hi = max(c.x.max() for c in curves)
    return lo, hi


def render(curves: list[Curve], title: str, out_stem: Path, box_text: str = "") -> None:
    margins = 1.2
    n_leg = max((len(c.label) for c in curves if c.label), default=0)
    n_box = max((len(t) for t in box_text.splitlines()), default=0)
    side_w = max(0.75 + 0.065 * n_leg, 0.45 + 0.065 * n_box, 1.4)

    fig = plt.figure(figsize=(8.2 + side_w + margins, 5.6))
    gs = fig.add_gridspec(1, 2, width_ratios=[8.2, side_w])
    ax = fig.add_subplot(gs[0])
    side = fig.add_subplot(gs[1])
    side.axis("off")

    for c in curves:
        ax.plot(c.x, c.y, color=c.color, lw=c.lw, alpha=c.alpha, zorder=3)
        if c.y_sd is not None and config.SHOW_SD_BAND:
            ax.fill_between(
                c.x, np.clip(c.y - c.y_sd, 0, None), c.y + c.y_sd,
                color=c.color, alpha=0.16, lw=0, zorder=2,
            )

    if config.SHOW_WAVELENGTH_MARKERS:
        for wl in [470, 520, 649, 664]:
            ax.axvline(x=wl, color="0.7", linestyle=":", lw=1.0, alpha=0.7, zorder=1)

    xlims = x_limits(curves)
    ax.set_xlim(*xlims)
    ax.set_ylim(-0.05, None)
    ax.set_xlabel(config.X_AXIS_LABEL, fontsize=11)
    ax.set_ylabel(config.Y_AXIS_LABEL, fontsize=11)
    if config.SHOW_GRID:
        ax.grid(True, linestyle="--", linewidth=0.65, alpha=0.42)
    ax.set_axisbelow(True)

    handles = [Line2D([], [], color=c.color, lw=c.lw, label=c.label) for c in curves if c.legend and c.label]
    legend = None
    if handles:
        legend = side.legend(handles=handles, loc="upper left", bbox_to_anchor=(0, 1),
                             bbox_transform=side.transAxes, borderaxespad=0,
                             frameon=True, framealpha=0.94, fontsize=8.5)

    if box_text and config.SHOW_STATS_BOX:
        y_top = 1.0
        if legend is not None:
            fig.canvas.draw()
            bbox = legend.get_window_extent().transformed(side.transAxes.inverted())
            y_top = bbox.y0 - 0.025
        shift = ScaledTranslation(0.055, 0, fig.dpi_scale_trans)
        box_style = {"boxstyle": "round,pad=0.45", "facecolor": "white", "edgecolor": "#B8C1CC", "alpha": 0.92}
        side.text(0, y_top, box_text, transform=side.transAxes + shift, ha="left", va="top", fontsize=8.5, bbox=box_style)

    fig.tight_layout()
    save_figure(fig, out_stem, config.PNG_DPI, config.SAVE_PDF)
    plt.close(fig)


def legend_label(spec: Spectrum) -> str:
    label = config.CUSTOM_LEGEND_LABELS.get(spec.name, spec.name)
    if config.STATS_IN_LEGEND and spec.stats.get("Chl_a_ug_mL"):
        label += f" (Chl a: {fmt(spec.stats['Chl_a_ug_mL'])} Âµg/mL)"
    return label


def plot_individual(spec: Spectrum, shade: str, out_dir: Path) -> None:
    folder = out_dir / safe_filename(spec.group or spec.name)
    curve = [Curve(spec.wavelength, spec.absorbance, legend_label(spec), shade)]
    title = config.CUSTOM_TITLES.get(spec.name, spec.name)
    render(curve, title, folder / f"{safe_filename(spec.name)}_spectrum", stats_text([spec]))


def plot_replicates(cond: Condition, family: str, out_dir: Path) -> None:
    colours = shades(family, len(cond.members))
    curves = [
        Curve(m.wavelength, m.absorbance, legend_label(m), colours[i])
        for i, m in enumerate(cond.members)
    ]
    render(curves, cond.name, out_dir / f"{safe_filename(cond.name)}_replicates", stats_text(cond.members))


def plot_comparison(key: tuple, conds: list[Condition], family: str, out_dir: Path) -> None:
    colours = shades(family, max(len(conds), 1))
    curves: list[Curve] = []
    for cond, colour in zip(conds, colours):
        x, mean_abs, sd_abs = common_curves(cond.members)
        multi = len(cond.members) > 1
        label = config.CUSTOM_LEGEND_LABELS.get(cond.name) or cond.name
        show_mean = config.COMPARE_SHOW_MEAN or not multi or not config.COMPARE_SHOW_REPLICATES

        if multi and config.COMPARE_SHOW_REPLICATES:
            for j, m in enumerate(cond.members):
                curves.append(Curve(
                    m.wavelength, m.absorbance,
                    label if (not show_mean and j == 0) else "",
                    colour,
                    lw=0.9 if show_mean else 1.8,
                    alpha=0.4 if show_mean else 0.9,
                    legend=(not show_mean and j == 0),
                ))
        if show_mean:
            curves.append(Curve(
                x, mean_abs, label, colour, lw=2.6,
                y_sd=sd_abs if multi else None,
            ))

    key_text = "_".join(str(v) for v in key if v is not None)
    title = config.CUSTOM_TITLES.get(key_text, f"{config.TITLE_PREFIX} â€” {key_text}")
    render(curves, title, out_dir / f"{safe_filename(key_text)}_comparison")


def plot_overview(conditions: list[Condition], family_for, out_dir: Path) -> None:
    n = len(conditions)
    if n == 0:
        return
    cols = min(4, n)
    rows = (n + cols - 1) // cols
    fig, axes = plt.subplots(rows, cols, figsize=(4.2 * cols, 3.0 * rows),
                             sharex=True, sharey=True, squeeze=False)
    axes = axes.ravel()
    for i, cond in enumerate(conditions):
        ax = axes[i]
        family = family_for(cond, i)
        colours = shades(family, max(len(cond.members), 1))
        for j, m in enumerate(cond.members):
            ax.plot(m.wavelength, m.absorbance, color=colours[j], lw=1.2)
        ax.set_title(cond.name, fontsize=8.5)
        if i % cols == 0:
            ax.set_ylabel(config.Y_AXIS_LABEL, fontsize=8.5)
        if i >= (rows - 1) * cols:
            ax.set_xlabel(config.X_AXIS_LABEL, fontsize=8.5)
        if config.SHOW_GRID:
            ax.grid(True, linestyle="--", linewidth=0.5, alpha=0.35)
    for j in range(n, len(axes)):
        axes[j].axis("off")
    fig.suptitle(f"{config.TITLE_PREFIX} â€” overview", y=1.02, fontsize=12, weight="semibold")
    fig.tight_layout()
    save_figure(fig, out_dir / "all_conditions", config.PNG_DPI, False)
    plt.close(fig)

def write_csvs(spectra: list[Spectrum], conditions: list[Condition], out: Path) -> None:
    out = Path(out)
    out.mkdir(parents=True, exist_ok=True)

    with (out / "spectrum_summary.csv").open("w", newline="", encoding="utf-8-sig") as f:
        w = csv.writer(f)
        header = [
            "name", "sample", "replicate", "source", "dilution_factor",
            "extract_volume_ml", "dry_mass_mg", "max_absorbance",
            "A_664", "A_649", "A_632", "A_520", "A_750",
            "Chl_a_ug_mL", "Chl_b_ug_mL", "Total_Chl_ug_mL",
            "Chl_a_ug_mg", "Chl_b_ug_mg", "Total_Chl_ug_mg", "flags"
        ]
        w.writerow(header)
        for s in spectra:
            w.writerow([
                s.name, s.group, s.replicate, s.source, s.dilution_factor,
                s.extract_volume_ml, s.dry_mass_mg, s.stats.get("max_absorbance"),
                s.stats.get("A_664"), s.stats.get("A_649"), s.stats.get("A_632"),
                s.stats.get("A_520"), s.stats.get("A_750"),
                s.stats.get("Chl_a_ug_mL"), s.stats.get("Chl_b_ug_mL"), s.stats.get("Total_Chl_ug_mL"),
                s.stats.get("Chl_a_ug_mg"), s.stats.get("Chl_b_ug_mg"), s.stats.get("Total_Chl_ug_mg"),
                "; ".join(s.flags),
            ])

    with (out / "condition_summary.csv").open("w", newline="", encoding="utf-8-sig") as f:
        w = csv.writer(f)
        metrics = ["max_absorbance", "A_664", "A_649", "A_632", "A_520", "A_750",
                   "Chl_a_ug_mL", "Chl_b_ug_mL", "Total_Chl_ug_mL",
                   "Chl_a_ug_mg", "Chl_b_ug_mg", "Total_Chl_ug_mg"]
        header = ["condition", "sample", "n"]
        for m in metrics:
            header += [f"{m}_mean", f"{m}_sd"]
        w.writerow(header)
        for cond in conditions:
            row = [cond.name, cond.sample, len(cond.members)]
            for m in metrics:
                v = stat_values(cond.members, m)
                row += [
                    v.mean() if v.size else None,
                    v.std(ddof=1) if v.size > 1 else None,
                ]
            w.writerow(row)

from __future__ import annotations

import string
from pathlib import Path
from typing import Optional, Union

from . import config
from .models import Condition, Spectrum
from .parsing import parse_spectrum_file, split_replicate
from .stats import calculate_pigments_and_metrics, palette_name, shades, stat_values
from .plotting import plot_comparison, plot_individual, plot_overview, plot_replicates
from .reports import write_csvs
from ..common.io import write_run_info
from ..common.plotting import apply_style


def _default_input_dir() -> Path:
    return Path.cwd()


def _collect_paths(in_dir: Path) -> list[Path]:
    paths = []
    if config.READ_TXT:
        paths += sorted(in_dir.glob("*.txt"), key=lambda p: p.name.lower())
    if config.READ_CSV:
        paths += sorted(in_dir.glob("*.csv"), key=lambda p: p.name.lower())
    return [p for p in paths if not p.name.lower().startswith(("requirements", "readme"))]


def organise(spectra: list[Spectrum]) -> list[Condition]:
    groups: dict[str, list[Spectrum]] = {}
    for spec in spectra:
        base, rep = split_replicate(spec.source)
        spec.group = base
        spec.replicate = rep
        spec.sample = base
        groups.setdefault(base, []).append(spec)

    conditions: list[Condition] = []
    for base, members in groups.items():
        used = {m.replicate for m in members if m.replicate}
        free = iter(c for c in string.ascii_lowercase if c not in used)
        for m in members:
            if not m.replicate:
                candidate = next(free, None)
                if candidate is None:
                    idx = sum(1 for x in members if m.replicate.startswith("rep"))
                    candidate = f"rep{idx + 1}"
                m.replicate = candidate
        members.sort(key=lambda m: m.replicate)

        for m in members:
            if len(members) > 1:
                m.name = f"{base} {m.replicate}"

        conditions.append(Condition(
            name=base,
            members=members,
            sample=base,
            dilution_factor=config.DEFAULT_DILUTION_FACTOR,
            extract_volume_ml=config.DEFAULT_EXTRACT_VOLUME_ML,
            dry_mass_mg=config.DEFAULT_DRY_MASS_MG,
        ))

    return sorted(conditions, key=lambda c: c.sample.lower())


def _family_for(cond: Condition, samples: list[str], fallback: int) -> str:
    for key in (cond.name, cond.sample):
        if key in config.GROUP_PALETTES:
            return config.GROUP_PALETTES[key]
    return palette_name(
        cond.sample,
        samples.index(cond.sample) if cond.sample in samples else fallback,
    )


def run(
    input_dir: Optional[Union[Path, str]] = None,
    output_dir: Optional[Union[Path, str]] = None,
) -> tuple[list[Spectrum], list[Condition]]:
    apply_style()

    in_dir = Path(input_dir).expanduser().resolve() if input_dir else _default_input_dir()
    out_dir = (Path(output_dir).expanduser().resolve() if output_dir
               else in_dir / config.OUTPUT_NAME)
    print(f"Reading UV-Vis from: {in_dir}")
    print(f"Writing to:          {out_dir}")

    spectra: list[Spectrum] = []
    for path in _collect_paths(in_dir):
        try:
            spec = parse_spectrum_file(path)
            spec.stats = calculate_pigments_and_metrics(spec)
            spectra.append(spec)
            print(f"[OK] {path.name}")
        except Exception as exc:
            print(f"[SKIP] {path.name}: {exc}")

    if not spectra:
        print(f"No usable UV-Vis files found in {in_dir}.")
        return [], []

    conditions = organise(spectra)
    samples = sorted({c.sample for c in conditions})

    for i, cond in enumerate(conditions):
        family = _family_for(cond, samples, i)
        if config.MAKE_INDIVIDUAL:
            colours = shades(family, max(len(cond.members), 1))
            for j, m in enumerate(cond.members):
                plot_individual(m, colours[j], out_dir / "individual")
        if config.MAKE_REPLICATES and len(cond.members) > 1:
            plot_replicates(cond, family, out_dir / "replicates")

    if config.MAKE_OVERVIEW:
        plot_overview(conditions, lambda c, i: _family_for(c, samples, i), out_dir / "overview")

    write_csvs(spectra, conditions, out_dir / "summary")
    write_run_info({
        "input_dir": str(in_dir),
        "output_dir": str(out_dir),
        "n_spectra": len(spectra),
        "n_conditions": len(conditions),
    }, out_dir)

    print("\nUV-Vis Quality Checks & Warnings:")
    all_warnings = [f"{s.name}: {w}" for s in spectra for w in s.flags]
    print("\n".join(f"  [WARN] {w}" for w in all_warnings) or "  none (all measurements passed)")
    print(f"\nResults folder: {out_dir}")
    return spectra, conditions

from __future__ import annotations

import numpy as np
from matplotlib.colors import LinearSegmentedColormap, to_hex

from . import config
from .models import Condition, Spectrum


def calculate_pigments_and_metrics(spec: Spectrum) -> dict:
    s = {}
    for wl in config.TARGET_WAVELENGTHS:
        s[f"A_{wl}"] = spec.get_absorbance_at(float(wl))

    a_750 = s.get("A_750", spec.get_absorbance_at(750.0))
    a_520 = s.get("A_520", spec.get_absorbance_at(520.0))
    a_649 = s.get("A_649", spec.get_absorbance_at(649.0))
    a_664 = s.get("A_664", spec.get_absorbance_at(664.0))

    red_region_mask = (spec.wavelength >= 630) & (spec.wavelength <= 680)
    red_max = float(spec.absorbance[red_region_mask].max()) if np.any(red_region_mask) else a_664

    max_abs = float(spec.absorbance.max()) if spec.absorbance.size > 0 else 0.0
    s["max_absorbance"] = max_abs

    flags = []
    if max_abs > config.MAX_ABSORBANCE_WARNING:
        flags.append(f"Max absorbance ({max_abs:.3f}) > {config.MAX_ABSORBANCE_WARNING}: Over 1.0, dilute and redo.")

    if abs(a_750) > config.TURBIDITY_THRESHOLD_A750:
        flags.append(f"A750 ({a_750:.3f}) > {config.TURBIDITY_THRESHOLD_A750}: Turbidity check failed (should be ~0).")

    if red_max > 0 and (a_520 / red_max) > config.CAROTENOID_MAX_RATIO_A520:
        flags.append(f"A520/RedMax ratio ({(a_520/red_max)*100:.1f}%) >= 10%.")

    spec.flags = flags
    s["flags"] = flags

    try:
        # Ritchie 2008 Ethanol Equations
        s["Chl_a_ug_mL"] = float(max(0.0, 13.36 * a_664 - 5.19 * a_649))
        s["Chl_b_ug_mL"] = float(max(0.0, 27.43 * a_649 - 8.12 * a_664))
        s["Total_Chl_ug_mL"] = s["Chl_a_ug_mL"] + s["Chl_b_ug_mL"]

        df = spec.dilution_factor
        v_etoh = spec.extract_volume_ml
        mass = spec.dry_mass_mg if spec.dry_mass_mg > 0 else 1.0

        # Chl (Âµg/mg d'extrait) = C (Âµg/mL) * facteur de dilution * V d'Ã©thanol de reprise (mL) / dry mass total (mg)
        s["Chl_a_ug_mg"] = (s["Chl_a_ug_mL"] * df * v_etoh) / mass
        s["Chl_b_ug_mg"] = (s["Chl_b_ug_mL"] * df * v_etoh) / mass
        s["Total_Chl_ug_mg"] = (s["Total_Chl_ug_mL"] * df * v_etoh) / mass

        s["Chl_a"] = s["Chl_a_ug_mg"]
        s["Chl_b"] = s["Chl_b_ug_mg"]
        s["Total_Chl"] = s["Total_Chl_ug_mg"]

    except Exception:
        s["Chl_a_ug_mL"] = 0.0
        s["Chl_b_ug_mL"] = 0.0
        s["Total_Chl_ug_mL"] = 0.0
        s["Chl_a_ug_mg"] = 0.0
        s["Chl_b_ug_mg"] = 0.0
        s["Total_Chl_ug_mg"] = 0.0
        s["Chl_a"] = 0.0
        s["Chl_b"] = 0.0
        s["Total_Chl"] = 0.0

    return s


def stat_values(members: list[Spectrum], key: str) -> np.ndarray:
    return np.array(
        [m.stats.get(key) for m in members if m.stats.get(key) is not None],
        dtype=float,
    )


def fmt(value: float) -> str:
    return f"{value:.3f}" if abs(value) >= 0.001 else f"{value:.2e}"


def stats_text(members: list[Spectrum]) -> str:
    lines = []
    for name in config.STATS_FIELDS:
        v = stat_values(members, name)
        if v.size == 0:
            continue
        if len(members) == 1:
            txt = fmt(v[0])
        else:
            txt = f"{fmt(v.mean())} Â± {fmt(v.std(ddof=1)) if v.size > 1 else 'â€“'}"
        lines.append(f"{name}: {txt}")
    if len(members) > 1:
        lines.append(f"(mean Â± SD, n = {len(members)})")
    return "\n".join(lines)


def palette_name(key, index: int) -> str:
    for candidate in (key, f"{key:g}" if isinstance(key, float) else None):
        if candidate in config.GROUP_PALETTES:
            return config.GROUP_PALETTES[candidate]
    if config.PALETTE_PRESET != "auto":
        return config.PALETTE_PRESET
    return config.PALETTE_CYCLE[index % len(config.PALETTE_CYCLE)]


def shades(name: str, n: int) -> list[str]:
    base = config.PALETTES[name]
    if n <= 1:
        return [base[1]]
    if n == 2:
        return [base[0], base[2]]
    if n == 3:
        return list(base)
    cmap = LinearSegmentedColormap.from_list(name, base)
    return [to_hex(cmap(x)) for x in np.linspace(0, 1, n)]


def common_curves(members: list[Spectrum]):
    x = members[0].wavelength
    abs_list = []
    for m in members:
        if m.wavelength.shape == x.shape and np.allclose(m.wavelength, x):
            abs_list.append(m.absorbance)
        else:
            abs_list.append(np.interp(x, m.wavelength, m.absorbance, left=0, right=0))
    arr = np.array(abs_list)
    sd = arr.std(axis=0, ddof=1) if len(members) > 1 else np.zeros(arr.shape[1])
    return x, arr.mean(axis=0), sd
